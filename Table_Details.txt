WITH TableSize AS
(
    SELECT
        t.object_id,
        t.name AS TableName,
        s.name AS SchemaName,
        SUM(p.rows) AS RowCount,
        CAST(SUM(a.total_pages) * 8.0 / 1024 AS DECIMAL(18,2)) AS TotalSizeMB
    FROM sys.tables t
    INNER JOIN sys.schemas s
        ON t.schema_id = s.schema_id
    INNER JOIN sys.indexes i
        ON t.object_id = i.object_id
    INNER JOIN sys.partitions p
        ON i.object_id = p.object_id
       AND i.index_id = p.index_id
    INNER JOIN sys.allocation_units a
        ON p.partition_id = a.container_id
    WHERE t.is_ms_shipped = 0
    GROUP BY t.object_id, t.name, s.name
)
SELECT
    ts.SchemaName,
    ts.TableName,
    ts.RowCount,
    ts.TotalSizeMB,

    -- Primary Keys
    PKs.PrimaryKeys,

    -- Foreign Keys
    FKs.ForeignKeys,

    -- Index Names
    IDX.IndexNames

FROM TableSize ts

OUTER APPLY
(
    SELECT STRING_AGG(k.name, ', ')
    FROM sys.key_constraints k
    WHERE k.parent_object_id = ts.object_id
      AND k.type = 'PK'
) PKs (PrimaryKeys)

OUTER APPLY
(
    SELECT STRING_AGG(fk.name, ', ')
    FROM sys.foreign_keys fk
    WHERE fk.parent_object_id = ts.object_id
) FKs (ForeignKeys)

OUTER APPLY
(
    SELECT STRING_AGG(i.name, ', ')
    FROM sys.indexes i
    WHERE i.object_id = ts.object_id
      AND i.name IS NOT NULL
      AND i.is_primary_key = 0
) IDX (IndexNames)

ORDER BY ts.TotalSizeMB DESC;
